{{- /* The ordered navigation of one product's docs, from the files alone: pages by their front matter weight; a subfolder (architecture/, packages/) is a group whose pages keep their own weights. Following the repos' convention, a group takes the first slot (10, 20, ...) the root pages leave free, in folder-name order; with none free it goes last. A folder inside a group with an _index.md (architecture/auth/) is a page of that group, by its _index.md's weight, whose own pages are nested under it by their weights. Returns a slice of (dict "weight" "page") or (dict "weight" "group" "pages" "children"), where children maps a nested section's .Path to its pages, by weight. */ -}} {{- $slug := . -}} {{- $root := site.GetPage (printf "/products/%s/docs" $slug) -}} {{- $entries := slice -}} {{- with $root -}} {{- $prefix := printf "products/%s/docs/" $slug -}} {{- $rootWeight := .Weight | default 10 -}} {{- $weights := slice $rootWeight -}} {{- $entries = $entries | append (dict "weight" $rootWeight "page" .) -}} {{- $groups := dict -}} {{- $children := dict -}} {{- $pages := .RegularPagesRecursive -}} {{- range .Sections -}}{{- $pages = $pages | append . -}}{{- end -}} {{- range sort (sort $pages "Title") "Weight" -}} {{- $pg := . -}} {{- $rel := strings.TrimPrefix $prefix .File.Dir -}} {{- /* a nested section sits in its parent folder */ -}} {{- if .IsSection -}}{{- $rel = strings.TrimSuffix "/" $rel | path.Dir | printf "%s/" | strings.TrimPrefix "./" -}}{{- end -}} {{- $parts := split (strings.TrimSuffix "/" $rel) "/" -}} {{- if eq $rel "" -}} {{- $weights = $weights | append .Weight -}} {{- $entries = $entries | append (dict "weight" .Weight "page" .) -}} {{- else if eq (len $parts) 1 -}} {{- $name := index $parts 0 -}} {{- $groups = merge $groups (dict $name ((index $groups $name | default (slice)) | append .)) -}} {{- else -}} {{- with .Parent -}} {{- $children = merge $children (dict .Path ((index $children .Path | default (slice)) | append $pg)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $next := 10 -}} {{- range $name, $pages := $groups -}} {{- /* first multiple of 10 no root page uses; past the last one if none is free */ -}} {{- range seq 1 100 -}}{{- if in $weights $next -}}{{ $next = add $next 10 }}{{- end -}}{{- end -}} {{- $weights = $weights | append $next -}} {{- $entries = $entries | append (dict "weight" $next "group" (humanize $name) "pages" $pages "children" $children) -}} {{- end -}} {{- end -}} {{- return (sort $entries "weight") -}}
